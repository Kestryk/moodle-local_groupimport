### Script reproductible des invariants de source

À exécuter avec Node depuis ce dépôt ; changer `base` uniquement pour comparer à une autre base. La syntaxe PHP et le test de sécurité se lancent séparément avec les commandes ci-dessus.

```javascript
const fs = require('node:fs');
const cp = require('node:child_process');
const assert = require('node:assert/strict');
const base = 'c2077524';
const assignment = /^(\$string\['([^']+)'\][ \t]*=[ \t]*)('(?:\\.|[^'\\])*'|"(?:\\.|[^"\\])*")([ \t]*;[^\r\n]*)(\r?)$/gm;
const tokens = value => value.match(/\{\$a(?:->[^{}]+)?\}|__[A-Za-z0-9_]+__|\{\}|\{[^{}\r\n]+\}|%(?:\d+\$)?[sdf]/g) || [];
for (const language of ['fr', 'en']) {
    const file = 'lang/' + language + '/local_groupimport.php';
    const before = cp.execFileSync('git', ['show', base + ':' + file], {encoding: 'utf8'});
    const after = fs.readFileSync(file, 'utf8');
    const left = [...before.matchAll(assignment)];
    const right = [...after.matchAll(assignment)];
    assert.equal(left.length, 1087);
    assert.equal(right.length, left.length);
    assert.deepEqual(right.map(row => row[2]), left.map(row => row[2]));
    const mask = source => source.replace(assignment,
        (_, prefix, key, value, suffix, cr) => prefix + 'VALUE' + suffix + cr);
    assert.equal(mask(after), mask(before));
    right.forEach((row, i) => assert.deepEqual(tokens(row[3]), tokens(left[i][3]), row[2]));
    console.log(language + ': counts, keys, placeholders, non-value bytes PASS');
}
```
