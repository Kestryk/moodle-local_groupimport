const fs = require('node:fs'),
      cp = require('node:child_process'),
      assert = require('node:assert/strict');

const {
  chromium
} = require('<PLAYWRIGHT_MODULE>');

const root = '<EASYSTUD_SOURCE_ROOT>';
const runId = 'codex1-k1680-f23c-native',
      out = __dirname;

const ps = (file, args) => cp.execFileSync('powershell.exe', ['-NoProfile', '-File', '<ORCHESTRATION_SCRIPTS>/' + file,
    ...args], {
  encoding: 'utf8'
});

const css = file => fs.readFileSync(file, 'utf8').replace(/url\((["']?)(\.\.?\/[^)"']+)\1\)/g, (match, quote, relative) =>
    'url("' + new URL(relative, "http://localhost:8051/local/groupimport/styles/student-manag" + "ement.css").href + '")');

async function settle(locator) {
  await locator.evaluate(async node => {
    const animations = [...node.getAnimations({
      subtree: true
    })];

    for (let n = node; n; n = n.parentElement) animations.push(...n.getAnimations());

    await Promise.all(animations.filter(a => Number.isFinite(a.effect.getTiming().iterations)).map(a => a.finished.catch((
        ) => {})));
  });
}

async function measure(locator) {
  return locator.evaluate(node => {
    const m = n => {
      const r = n.getBoundingClientRect(),
            s = getComputedStyle(n);
      return {
        width: r.width,
        height: r.height,
        padding: s.padding,
        radius: s.borderRadius,
        color: s.color,
        background: s.backgroundColor,
        border: s.border,
        fontSize: s.fontSize,
        fontWeight: s.fontWeight,
        lineHeight: s.lineHeight,
        overflow: s.overflow,
        textOverflow: s.textOverflow
      };
    };

    const selectors = ['.local-groupimport-easystud-settings-modal__image',
        ".local-groupimport-easystud-settings-modal__image-placeholde" + "r",
            ".local-groupimport-easystud-settings-modal__image-placeholde" + "r > .fa", '[data-easystud-advanced-file-name]'];
    return selectors.map(selector => {
      const n = node.querySelector(selector);
      if (!n) throw Error('Missing native preview ' + selector);
      let glyph;

      if (n.matches('.fa')) {
        const b = n.getBoundingClientRect(),
              raw = getComputedStyle(n, '::before').content;
        const t = document.createTextNode(raw.slice(1, -1));
        n.classList.add('codex1-range-probe');
        n.append(t);

        try {
          const range = document.createRange();
          range.selectNodeContents(t);
          const g = range.getBoundingClientRect(),
                a = n.getBoundingClientRect();
          glyph = {
            sameFrame: b.width === a.width && b.height === a.height,
            x: g.x + g.width / 2 - b.x - b.width / 2,
            y: g.y + g.height / 2 - b.y - b.height / 2
          };
        } finally {
          t.remove();
          n.classList.remove('codex1-range-probe');
        }
      }

      return {
        selector,
        visible: Boolean(n.getClientRects().length),
        ...m(n),
        glyph
      };
    });
  });
}

(async () => {
  const result = {
    passed: false,
    rows: [],
    errors: [],
    runtimeFilesChanged: false
  };
  let lease = false,
      created = false,
      browser,
      page;

  try {
    ps('Acquire-EasyEduResourceLease.ps1', ['-Resource', 'moodle51-perf', '-ProjectNamespace', 'easystud', '-RunId', runId,
        '-Repository', root, '-Purpose', "K16-80 disclosures exact native CSS parity, restored synthet" + "ic fixture",
            '-LeaseSeconds', '900', '-OwnerPid', String(process.pid)]);
    lease = true;
    const fixture = JSON.parse(cp.execFileSync('<PHP_BINARY>', [out + '/fixture.php', 'create'], {
      encoding: 'utf8'
    }));
    created = true;
    result.course = fixture.courseid;
    browser = await chromium.launch({
      headless: true,
      executablePath: '<CHROME_EXECUTABLE>'
    });

    for (const locale of ['fr', 'en']) for (const width of [1600, 390]) {
      const context = await browser.newContext({
        storageState: '<PROCESS_LOCAL_AUTH_STATE>',
        viewport: {
          width,
          height: 1000
        },
        hasTouch: width === 390,
        reducedMotion: 'reduce'
      });
      page = await context.newPage();
      page.setDefaultTimeout(20000);
      page.on('pageerror', e => result.errors.push(e.message));
      page.on('console', m => {
        if (m.type() === 'error') result.errors.push(m.text());
      });
      const response = await page.goto('http://localhost:8051/local/groupimport/manage.php?id=' + fixture.courseid +
          '&lang=' + locale, {
        waitUntil: 'domcontentloaded',
        timeout: 180000
      });
      assert.equal(response.status(), 200);
      await page.waitForFunction(() => {
        const r = document.querySelector('#local-groupimport-easystud');
        return r?.dataset.easystudManagerInitialised === '1' && !r.classList.contains('is-action-busy') &&
            !r.classList.contains('local-groupimport-easystud--booting');
      }, null, {
        timeout: 180000
      });
      const both = page.locator('[data-easystud-layout-mode="both"]');
      if (await both.isVisible()) await both.click();
      if (width < 1024) await page.locator('[data-easystud-mobile-view="groups"]').click();
      await page.evaluate(() => {
        document.querySelectorAll('link[rel="stylesheet"]').forEach(n => {
          if (/\/local\/groupimport\/styles\//.test(n.href)) n.disabled = true;
        });
      });
      const style = await page.addStyleTag({
        content: css(out + '/before-student-management.css')
      });
      await page.addStyleTag({
        content: '.codex1-range-probe::before {content:none!important;}'
      });
      await page.evaluate(() => document.fonts.ready);
      const ungrouped = page.locator('[aria-controls="easystud-group-list-ungrouped"]');
      if ((await ungrouped.isVisible()) && (await ungrouped.getAttribute('aria-expanded')) === 'false') await ungrouped.click();
      const card = page.locator("[data-selectable-type=\"group\"][data-easystud-group-id]:visib" + "le").first();
      const direct = card.locator(":scope > .local-groupimport-easystud-group__header > [data-e" +
          "asystud-open-advanced-settings]");
      if (await direct.isVisible()) await direct.click();else {
        await card.locator(":scope > .local-groupimport-easystud-group__header .local-gr" +
            "oupimport-easystud-group__name").click({
          button: 'right'
        });
        await page.locator("[data-easystud-context-action=\"group-open-advanced-settings\"" + "]:visible").click();
      }
      const modal = page.locator('[data-easystud-advanced-settings-modal]:visible');
      await modal.waitFor({
        state: 'visible'
      });
      await modal.locator('[data-easystud-settings-list-section]').first().waitFor({
        state: 'visible'
      });
      await page.waitForFunction(() => {
        const modal = document.querySelector('[data-easystud-advanced-settings-modal]:not([hidden])');
        return modal?.querySelector('button[type="submit"]')?.disabled === false;
      });

      for (const phase of ['before', 'after']) {
        await style.evaluate((n, value) => {
          n.textContent = value;
        }, css(phase === 'before' ? out + '/before-student-management.css' : root + '/styles/student-management.css'));
        await settle(modal);
        result.rows.push({
          kind: 'dialog-shell',
          locale,
          width,
          phase,
          measurement: await measure(modal)
        });
      }

      await context.close();
    }

    assert.equal(result.rows.length, 8);

    for (const after of result.rows.filter(r => r.phase === 'after')) {
      const before = result.rows.find(r => r.locale === after.locale && r.width === after.width && r.phase === 'before');
      const a = structuredClone(after.measurement),
            b = structuredClone(before.measurement);

      for (const rows of [a, b]) for (const part of rows) {
        if (part.glyph) {
          delete part.glyph;
          delete part.lineHeight;
        }
      }

      assert.deepEqual(a, b, "Exact native preview sizes, type, padding, radii and paint o" + "utside intended glyph metrics");
      assert.equal(after.measurement.length, 4);
      assert.ok(after.measurement.every(x => x.visible));

      for (const part of after.measurement) if (part.glyph) assert.ok(part.glyph.sameFrame && Math.abs(
          part.glyph.x) <= 0.5 && Math.abs(part.glyph.y) <= 0.5, 'Preview glyph centre within 0.5px');
    }

    assert.equal(result.errors.length, 0);
    result.passed = true;
  } catch (e) {
    result.failure = e.message;
    if (page && !page.isClosed()) await page.screenshot({
      path: out + '/failure.png',
      timeout: 10000
    }).catch(() => {});
  } finally {
    if (browser) await browser.close();

    if (!created && fs.existsSync(out + '/fixture-ids.json')) {
      const ids = JSON.parse(fs.readFileSync(out + '/fixture-ids.json'));
      const restored = fs.existsSync(out + '/fixture-restored.json') ? JSON.parse(fs.readFileSync(out +
          '/fixture-restored.json')) : {};
      created = !(restored.removed && restored.courseid === ids.courseid);
    }

    if (created) result.fixtureCleanup = JSON.parse(cp.execFileSync('<PHP_BINARY>', [out + '/fixture.php', 'cleanup'], {
      encoding: 'utf8'
    }));

    if (lease) {
      ps('Release-EasyEduResourceLease.ps1', ['-Resource', 'moodle51-perf', '-RunId', runId]);
      result.leaseReleased = true;
    }

    fs.writeFileSync(out + '/native.json', JSON.stringify(result, null, 2));
    console.log(JSON.stringify({
      passed: result.passed,
      rows: result.rows.length,
      errors: result.errors.length,
      failure: result.failure,
      restored: result.fixtureCleanup,
      leaseReleased: result.leaseReleased
    }));
  }
})();
