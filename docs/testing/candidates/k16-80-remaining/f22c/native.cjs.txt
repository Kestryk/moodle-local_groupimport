async function settle(locator) {
    await locator.evaluate(async node => {
        const animations = [...node.getAnimations({subtree:true})];
        for (let n = node; n; n = n.parentElement) animations.push(...n.getAnimations());
        await Promise.all(animations.filter(a => Number.isFinite(a.effect.getTiming().iterations)).map(a => a.finished.catch(() => {})));
    });
}
async function measure(locator) {
    return locator.evaluate(node => {
        const r = node.getBoundingClientRect(), s = getComputedStyle(node);
        const textRects=[];const walker=document.createTreeWalker(node,NodeFilter.SHOW_TEXT);
        while(walker.nextNode()) {if(!walker.currentNode.textContent.trim())continue;const range=document.createRange();range.selectNodeContents(walker.currentNode);for(const b of range.getClientRects())if(b.width)textRects.push(b);}
        const top=Math.min(...textRects.map(b=>b.y)),bottom=Math.max(...textRects.map(b=>b.bottom));
        const iconRects=[...node.querySelectorAll('.fa')].map(icon=>icon.getBoundingClientRect());
        const allRects=[...textRects,...iconRects],left=Math.min(...allRects.map(b=>b.x)),right=Math.max(...allRects.map(b=>b.right));
        const groupTop=Math.min(...allRects.map(b=>b.y)),groupBottom=Math.max(...allRects.map(b=>b.bottom));
        const glyphs=[...node.querySelectorAll('.fa')].map(icon=>{
            const b=icon.getBoundingClientRect(),raw=getComputedStyle(icon,'::before').content;
            const t=document.createTextNode(raw.slice(1,-1));icon.classList.add('codex1-range-probe');icon.append(t);
            try{
                const range=document.createRange();range.selectNodeContents(t);const g=range.getBoundingClientRect(),after=icon.getBoundingClientRect();
                return {width:b.width,height:b.height,centerY:b.y+b.height/2-r.y-r.height/2,probeSameFrame:after.width===b.width&&after.height===b.height,
                    glyphCenterX:g.x+g.width/2-b.x-b.width/2,glyphCenterY:g.y+g.height/2-b.y-b.height/2};
            }finally{t.remove();icon.classList.remove('codex1-range-probe');}
        });
        return {width:r.width,height:r.height,padding:s.padding,radius:s.borderRadius,color:s.color,background:s.backgroundColor,
            border:s.border,lineHeight:s.lineHeight,fontSize:s.fontSize,fontWeight:s.fontWeight,
            centerY:(top+bottom)/2-r.y-r.height/2,contentCenterX:(left+right)/2-r.x-r.width/2,contentCenterY:(groupTop+groupBottom)/2-r.y-r.height/2,glyphs};
    });
}
