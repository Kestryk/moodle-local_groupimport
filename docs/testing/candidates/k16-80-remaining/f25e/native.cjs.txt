const fs = require('node:fs'), cp = require('node:child_process'), assert = require('node:assert/strict');
const {chromium} = require('<PLAYWRIGHT_MODULE>');
const root = '<EASYSTUD_SOURCE_ROOT>';
const runId = 'codex1-k1680-f25e-native', out = __dirname;
const ps = (file, args) => cp.execFileSync('powershell.exe', ['-NoProfile', '-File', '<ORCHESTRATION_SCRIPTS>/' + file, ...args], {encoding: 'utf8'});
const css = file => fs.readFileSync(file, 'utf8').replace(/url\((["']?)(\.\.?\/[^)"']+)\1\)/g,
    (match, quote, relative) => 'url("' + new URL(relative, 'http://localhost:8051/local/groupimport/styles/student-management.css').href + '")');
async function settle(locator) {
    await locator.evaluate(async node => {
        const animations = [...node.getAnimations({subtree:true})];
        for (let n = node; n; n = n.parentElement) animations.push(...n.getAnimations());
        await Promise.all(animations.filter(a => Number.isFinite(a.effect.getTiming().iterations)).map(a => a.finished.catch(() => {})));
    });
}
async function measure(locator,time) {
 await locator.evaluate((n,time)=>{for(const a of n.getAnimations({subtree:true})){a.pause();a.currentTime=time;}},time);
 return locator.evaluate(node=>[...node.querySelectorAll('.local-groupimport-easystud__loading-participant-card,.local-groupimport-easystud__loading-structure-card,.local-groupimport-easystud__loading-card-avatar,.local-groupimport-easystud__loading-card-line')].filter(n=>n.getClientRects().length).map(n=>{const b=n.getBoundingClientRect(),s=getComputedStyle(n);return {class:n.className,width:b.width,height:b.height,blockSize:s.blockSize,padding:s.padding,radius:s.borderRadius,background:s.background,border:s.border,animation:s.animationName};}));
}
(async () => {
    const result = {passed: false, rows: [], errors: [], runtimeFilesChanged: false};
    let lease = false, created = false, browser, page,releaseRequest;
    try {
        ps('Acquire-EasyEduResourceLease.ps1', ['-Resource', 'moodle51-perf', '-ProjectNamespace', 'easystud', '-RunId', runId,
            '-Repository', root, '-Purpose', 'K16-80 disclosures exact native CSS parity, restored synthetic fixture', '-LeaseSeconds', '900', '-OwnerPid', String(process.pid)]);
        lease = true;
        const fixture = JSON.parse(cp.execFileSync('<PHP_BINARY>', [out + '/fixture.php', 'create'], {encoding: 'utf8'}));
        created = true; result.course = fixture.courseid;
        browser = await chromium.launch({headless: true, executablePath: '<CHROME_EXECUTABLE>'});
        for (const locale of ['fr', 'en']) for (const width of [1600, 390]) {
            const context = await browser.newContext({storageState: '<PROCESS_LOCAL_AUTH_STATE>',
                viewport: {width, height: 1000}, hasTouch: width === 390, reducedMotion: 'reduce'});
            page = await context.newPage();
            page.setDefaultTimeout(20000);
            page.on('pageerror', e => result.errors.push(e.message));
            page.on('console', m => {if (m.type() === 'error') result.errors.push(m.text());});
            let held=0;const gate=new Promise(resolve=>{releaseRequest=resolve;});
            await page.route('**/lib/ajax/service.php*',async route=>{
                const payload=(route.request().postData()||'')+decodeURIComponent(route.request().url());
                if(payload.includes('local_groupimport_get_state')){held++;await gate;}
                await route.continue().catch(()=>{});
            });
            const response = await page.goto('http://localhost:8051/local/groupimport/manage.php?id=' + fixture.courseid + '&lang=' + locale,
                {waitUntil: 'domcontentloaded', timeout: 180000});
            assert.equal(response.status(), 200);
            const retired=".local-groupimport-easystud__loading-card-copy,.local-groupimport-easystud__loading-filter-cluster,.local-groupimport-easystud__loading-pagination,.local-groupimport-easystud__loading-pagination-rail,.local-groupimport-easystud__loading-pagination-summary,.local-groupimport-easystud__loading-panel,.local-groupimport-easystud__loading-panel--participants,.local-groupimport-easystud__loading-panel--structure,.local-groupimport-easystud__loading-panel-header,.local-groupimport-easystud__loading-panel-secondary,.local-groupimport-easystud__loading-panel-title,.local-groupimport-easystud__loading-search,.local-groupimport-easystud__loading-search-filter-region,.local-groupimport-easystud__loading-structure-tools,.local-groupimport-easystud__loading-view-toggle,.local-groupimport-easystud__loading-view-toggle--two,.local-groupimport-easystud__loading-view-toggle-item";assert.equal(await page.locator(retired).count(),0,'No retired fake-page hosts during real component loading');const participants=page.locator('[data-easystud-sm-loading="participants"]:visible').first();
            await participants.waitFor({state:'visible'});
            await page.evaluate(()=>{document.querySelectorAll('link[rel="stylesheet"]').forEach(n=>{if(/\/local\/groupimport\/styles\//.test(n.href))n.disabled=true;});});
            const style=await page.addStyleTag({content:css(out+'/before-student-management.css')});
            await page.evaluate(()=>document.fonts.ready);
            const targets=[{kind:'participant-loading',locator:participants}];
            const groups=page.locator('[data-easystud-sm-loading="groups"]:visible').first();
            if(await groups.count())targets.push({kind:'group-loading',locator:groups});
            result.coverage??=[];result.coverage.push({locale,width,targets:targets.map(t=>t.kind)});
            for(const phase of ['before','after'])for(const time of [0,400])for(const target of targets){
                await style.evaluate((n,t)=>{n.textContent=t;},css(phase==='before'?out+'/before-student-management.css':root+'/styles/student-management.css'));
                result.rows.push({kind:target.kind,locale,width,time,phase,measurement:await measure(target.locator,time)});
            }
            releaseRequest();releaseRequest=null;
            await page.waitForFunction(()=>document.querySelector('#local-groupimport-easystud')?.dataset.easystudManagerInitialised==='1',null,{timeout:180000});
            assert.ok(held>0,'Actual get_state request deferred');assert.equal(await page.locator(retired).count(),0,'No retired fake-page hosts after readiness');
            result.readyChecks??=[];result.readyChecks.push({locale,width,ready:true});
            await context.close();
        }
        assert.ok(result.rows.length>=16);assert.equal(result.readyChecks.length,4);
        for(const after of result.rows.filter(r=>r.phase==='after')){
            const before=result.rows.find(r=>r.kind===after.kind&&r.locale===after.locale&&r.width===after.width&&r.time===after.time&&r.phase==='before');

        }

    } catch (e) {
        result.failure = e.message;
        if (page && !page.isClosed()) await page.screenshot({path: out + '/failure.png', timeout: 10000}).catch(() => {});

    }
